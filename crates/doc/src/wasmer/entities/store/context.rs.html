1//! Thread-local storage for storing the current store context,
2//! i.e. the currently active [`Store`](crate::Store)(s). When
3//! a function is called, a pointer to the [`StoreInner`] in placed
4//! inside the store context so it can be retrieved when needed.
5//! This lets code that needs access to the store get it with
6//! just the store ID.
7//!
8//! The currently active store context can be a sync or async
9//! context.
10//!
11//! For sync contexts, we just store a raw pointer
12//! to the `StoreInner`, which is owned by the embedder's stack.
13//!
14//! For async contexts, we store a write guard taken from the
15//! [`StoreAsync`](crate::StoreAsync); This achieves two goals:
16//!   * Makes the [`StoreAsync`](crate::StoreAsync) available
17//!     to whoever needs it, including when code needs to spawn
18//!     new coroutines
19//!   * Makes sure a write lock is held on the store as long as
20//!     the context is active, preventing other tasks from
21//!     accessing the store concurrently.
22//!
23//! We maintain a stack because it is technically possible to
24//! have nested `Function::call` invocations that use different
25//! stores, such as:
26//!     call(store1, func1) -> wasm code -> imported func ->
27//!     call(store2, func2)
28//!
29//! Note that this stack is maintained by both function
30//! calls and the async_runtime to reflect the exact WASM
31//! functions running on a given thread at any moment in
32//! time. If a function suspends, its store context is
33//! cleared and later reinstalled when it resumes. This lets
34//! us use thread-local storage for the context without
35//! requiring that async tasks are tied to specific threads.
36//!
37//! When something needs the "currently active" store context,
38//! they will only look at the top entry in the stack. It is
39//! always an error for code to try to access a store that's
40//! "paused", i.e. not the top entry. This should be impossible
41//! due to how the function call code is structured, but we
42//! guard against it anyway.
43

58enum StoreContextEntry {
59    Sync(*mut StoreInner),
60
61    #[cfg(feature = "experimental-async")]
62    Async(LocalRwLockWriteGuard<Box<StoreInner>>),
63}

65impl StoreContextEntry {
66    fn as_ptr(&self) -> *mut StoreInner {
67        match self {
68            Self::Sync(ptr) => *ptr,
69            #[cfg(feature = "experimental-async")]
70            Self::Async(guard) => &***guard as *const _ as *mut _,
71        }
72    }
73}

78    // StoreContexts can be used recursively when Function::call
79    // is used in an imported function. In the scenario, we're
80    // essentially passing a mutable borrow of the store into
81    // Function::call. However, entering the WASM code loses the
82    // reference, and it needs to be re-acquired from the
83    // StoreContext. This is why we use an UnsafeCell to allow
84    // multiple mutable references to the StoreMut; we do however
85    // keep track of how many borrows there are so we don't drop
86    // it prematurely.

88    entry: UnsafeCell<StoreContextEntry>,
89}

117pub(crate) struct StoreInstallGuard {
118    /// `None` when this guard installed nothing, which happens only for a
119    /// store an async context already holds — see [`StoreContext::install`].
120    store_id: Option<StoreId>,
121}

123thread_local! {
124    static STORE_CONTEXT_STACK: RefCell<Vec<StoreContext>> = const { RefCell::new(Vec::new()) };
125}

127impl StoreContext {
128    fn is_active(id: StoreId) -> bool {
129        STORE_CONTEXT_STACK.with(|cell| {
130            let stack = cell.borrow();
131            stack.last().is_some_and(|ctx| ctx.id == id)
132        })
133    }
134
135    fn is_suspended(id: StoreId) -> bool {
136        !Self::is_active(id)
137            && STORE_CONTEXT_STACK.with(|cell| {
138                let stack = cell.borrow();
139                stack.iter().rev().skip(1).any(|ctx| ctx.id == id)
140            })
141    }
142
143    fn push(id: StoreId, entry: StoreContextEntry) {
144        STORE_CONTEXT_STACK.with(|cell| {
145            let mut stack = cell.borrow_mut();
146            stack.push(Self {
147                id,
148                borrow_count: 0,
149                entry: UnsafeCell::new(entry),
150            });
151        })
152    }
153
154    #[cfg(feature = "unsafe-cothread")]
155    fn push_cothread(id: StoreId, store_ptr: NonNull<StoreInner>) {
156        STORE_CONTEXT_STACK.with(|cell| {
157            let mut stack = cell.borrow_mut();
158            stack.push(Self {
159                id,
160                borrow_count: 1,
161                entry: UnsafeCell::new(StoreContextEntry::Sync(store_ptr.as_ptr())),
162            });
163        });
164    }
165
166    #[cfg(feature = "unsafe-cothread")]
167    fn uninstall_cothread(id: StoreId) {
168        STORE_CONTEXT_STACK.with(|cell| {
169            let mut stack = cell.borrow_mut();
170            // Search from the top so we find our entry even if the stack has
171            // been disturbed (e.g. another guard panicked mid-flight).
172            if let Some(pos) = stack.iter().rposition(|ctx| ctx.id == id) {
173                stack.remove(pos);
174            } else {
175                panic!(
176                    "CoroutineStoreGuard::drop: entry not found in context stack; \
177                        the store context stack is corrupted"
178                );
179            }
180        });
181    }
182
183    /// Returns true if there are no active store context entries.
184    pub(crate) fn is_empty() -> bool {
185        STORE_CONTEXT_STACK.with(|cell| {
186            let stack = cell.borrow();
187            stack.is_empty()
188        })
189    }
190
191    /// The write guard ensures this is the only reference to the store,
192    /// so installation can never fail.
193    #[cfg(feature = "experimental-async")]
194    pub(crate) fn install_async(
195        guard: LocalRwLockWriteGuard<Box<StoreInner>>,
196    ) -> ForcedStoreInstallGuard {
197        let store_id = guard.objects.id();
198        Self::push(store_id, StoreContextEntry::Async(guard));
199        ForcedStoreInstallGuard { store_id }
200    }
201
202    /// Whether the active entry is `id`'s, held by an async context.
203    fn active_is_async(id: StoreId) -> bool {
204        STORE_CONTEXT_STACK.with(|cell| {
205            let stack = cell.borrow();
206            let Some(top) = stack.last() else {
207                return false;
208            };
209            if top.id != id {
210                return false;
211            }
212            match unsafe { top.entry.get().as_ref().unwrap() } {
213                StoreContextEntry::Sync(_) => false,
214                #[cfg(feature = "experimental-async")]
215                StoreContextEntry::Async(_) => true,
216            }
217        })
218    }
219
220    /// Install `store_ptr` as this thread's executing store until the returned
221    /// guard is dropped.
222    ///
223    /// A store already executing here gets a *second* entry rather than
224    /// re-using its first: the entries differ in which borrow their pointer was
225    /// derived from, and that difference is load-bearing. Every re-acquisition
226    /// ([`Self::get_current`] and friends) reborrows the top entry's pointer,
227    /// so re-using an outer entry would make the inner borrow a *sibling* of
228    /// the one the caller is still holding — creating it invalidates the
229    /// caller's, and the caller's next use of its own store is undefined
230    /// behaviour. Installing the pointer the caller just derived from its live
231    /// borrow makes the inner borrow a child of it instead, which is what lets
232    /// the caller carry on once the nested call returns. See the
233    /// `borrow_provenance` tests, which fail under Miri if this re-uses the
234    /// outer entry.
235    ///
236    /// An async context is the exception, and installs nothing: it owns the
237    /// store through a write guard held in the entry itself, and every
238    /// re-acquisition re-derives from that guard rather than from a caller's
239    /// borrow (see `AsyncCallStoreMut`), so there is no borrow to nest under.
240    /// Shadowing it would also hide the guard from async host functions, which
241    /// reach it by inspecting the active entry.
242    ///
243    /// # Safety
244    /// `store_ptr` must be derived from a mutable borrow of the store that
245    /// stays alive, and unreachable to every frame on this thread, until the
246    /// guard is dropped.
247    pub(crate) unsafe fn install(store_ptr: *mut StoreInner) -> StoreInstallGuard {
248        let store_id = unsafe { store_ptr.as_ref().unwrap().objects.id() };
249        // Nesting changes a pointer's provenance, never which store it
250        // addresses, so the same store must arrive at the same address.
251        debug_assert!(
252            !Self::is_active(store_id)
253                || STORE_CONTEXT_STACK.with(|cell| {
254                    let stack = cell.borrow();
255                    let active =
256                        unsafe { stack.last().unwrap().entry.get().as_ref().unwrap().as_ptr() };
257                    active == store_ptr
258                }),
259            "Store context pointer mismatch"
260        );
261        if Self::active_is_async(store_id) {
262            return StoreInstallGuard { store_id: None };
263        }
264        Self::push(store_id, StoreContextEntry::Sync(store_ptr));
265        StoreInstallGuard {
266            store_id: Some(store_id),
267        }
268    }
269
270    /// "Pause" one borrow of the store context.
271    ///
272    /// # Safety
273    /// Code must ensure it does not use the StorePtrWrapper or
274    /// StoreAsyncGuardWrapper that it owns, or any StoreRef/StoreMut
275    /// derived from them, while the store context is paused.
276    ///
277    /// The safe, correct use-case for this method is to
278    /// pause the store context while executing WASM code, which
279    /// cannot use the store context directly. This allows an async
280    /// context to uninstall the store context when suspending if it's
281    /// called from a sync imported function. The imported function
282    /// will have borrowed the store context in its trampoline, which
283    /// will prevent the async context from uninstalling the store.
284    /// However, since the imported function passes a mutable borrow
285    /// of its store into `Function::call`, it will expect the store
286    /// to change before the call returns.
287    pub(crate) unsafe fn pause(id: StoreId) -> StorePtrPauseGuard {
288        STORE_CONTEXT_STACK.with(|cell| {
289            let mut stack = cell.borrow_mut();
290            let top = stack
291                .last_mut()
292                .expect("No store context installed on this thread");
293            assert_eq!(top.id, id, "Mismatched store context access");
294            let ref_count_decremented = if top.borrow_count > 0 {
295                top.borrow_count -= 1;
296                true
297            } else {
298                false
299            };
300            StorePtrPauseGuard {
301                store_id: id,
302                ptr: unsafe { top.entry.get().as_ref().unwrap().as_ptr() },
303                ref_count_decremented,
304            }
305        })
306    }
307
308    /// The active entry's store, if no frame on this thread's stack is holding
309    /// a borrow of it.
310    ///
311    /// A non-zero borrow count means some frame further up still holds a
312    /// `StoreMut` it can reach, so handing out a second one would alias it;
313    /// the caller gets `None` instead. A frame that has lent its borrow out
314    /// ([`StoreMut::parked`]) installed an entry of its own, whose count is
315    /// zero until somebody takes it — which is how an imported function lends
316    /// its store to code it calls into.
317    pub(crate) fn try_get_current_unborrowed() -> Option<StorePtrWrapper> {
318        STORE_CONTEXT_STACK.with(|cell| {
319            let mut stack = cell.borrow_mut();
320            let top = stack.last_mut()?;
321            if top.borrow_count != 0 {
322                return None;
323            }
324            top.borrow_count += 1;
325            Some(StorePtrWrapper {
326                store_ptr: unsafe { top.entry.get().as_mut().unwrap().as_ptr() },
327            })
328        })
329    }
330
331    /// Safety: This method lets you borrow multiple mutable references
332    /// to the currently active store context. The caller must ensure that:
333    ///   * there is only one mutable reference alive, or
334    ///   * all but one mutable reference are inaccessible and passed
335    ///     into a function that lost the reference (e.g. into WASM code)
336    ///
337    /// The intended, valid use-case for this method is from within
338    /// imported function trampolines.
339    pub(crate) unsafe fn get_current(id: StoreId) -> StorePtrWrapper {
340        STORE_CONTEXT_STACK.with(|cell| {
341            let mut stack = cell.borrow_mut();
342            let top = stack
343                .last_mut()
344                .expect("No store context installed on this thread");
345            assert_eq!(top.id, id, "Mismatched store context access");
346            top.borrow_count += 1;
347            StorePtrWrapper {
348                store_ptr: unsafe { top.entry.get().as_mut().unwrap().as_ptr() },
349            }
350        })
351    }
352
353    /// Safety: In addition to the safety requirements of [`Self::get_current`],
354    /// the pointer returned from this function will become invalid if
355    /// the store context is changed in any way (via installing or uninstalling
356    /// a store context). The caller must ensure that the store context
357    /// remains unchanged as long as the pointer is being accessed.
358    pub(crate) unsafe fn get_current_transient(id: StoreId) -> *mut StoreInner {
359        STORE_CONTEXT_STACK.with(|cell| {
360            let mut stack = cell.borrow_mut();
361            let top = stack
362                .last_mut()
363                .expect("No store context installed on this thread");
364            assert_eq!(top.id, id, "Mismatched store context access");
365            unsafe { top.entry.get().as_mut().unwrap().as_ptr() }
366        })
367    }
368
369    /// Safety: See [`Self::get_current`].
370    pub(crate) unsafe fn try_get_current(id: StoreId) -> Option<StorePtrWrapper> {
371        STORE_CONTEXT_STACK.with(|cell| {
372            let mut stack = cell.borrow_mut();
373            let top = stack.last_mut()?;
374            if top.id != id {
375                return None;
376            }
377            top.borrow_count += 1;
378            Some(StorePtrWrapper {
379                store_ptr: unsafe { top.entry.get().as_mut().unwrap().as_ptr() },
380            })
381        })
382    }
383
384    /// Safety: See [`Self::get_current`].
385    #[cfg(feature = "experimental-async")]
386    pub(crate) unsafe fn try_get_current_async(id: StoreId) -> GetStoreAsyncGuardResult {
387        STORE_CONTEXT_STACK.with(|cell| {
388            let mut stack = cell.borrow_mut();
389            let Some(top) = stack.last_mut() else {
390                return GetStoreAsyncGuardResult::NotInstalled;
391            };
392            if top.id != id {
393                return GetStoreAsyncGuardResult::NotInstalled;
394            }
395            top.borrow_count += 1;
396            match unsafe { top.entry.get().as_mut().unwrap() } {
397                StoreContextEntry::Async(guard) => {
398                    GetStoreAsyncGuardResult::Ok(StoreAsyncGuardWrapper {
399                        guard: guard as *mut _,
400                    })
401                }
402                StoreContextEntry::Sync(ptr) => {
403                    GetStoreAsyncGuardResult::NotAsync(StorePtrWrapper { store_ptr: *ptr })
404                }
405            }
406        })
407    }

410#[cfg(feature = "unsafe-cothread")]
411/// RAII guard that installs the store context on the thread-local stack when
412/// created and removes it on drop. See [`crate::Store::coroutine_store_guard`].
413pub struct CoroutineStoreGuard<'a> {
414    store_id: StoreId,
415    _store: std::marker::PhantomData<&'a mut StoreInner>,
416}
417
418#[cfg(feature = "unsafe-cothread")]
419impl<'a> CoroutineStoreGuard<'a> {
420    /// # Panics
421    /// Panics if the store is anywhere on the current thread's context stack
422    /// (active or suspended).
423    ///
424    /// # Safety
425    /// Exactly one `StorePtrWrapper` derived from `store` must be alive on
426    /// the suspended coroutine's stack for the duration of the guard.
427    pub(crate) unsafe fn new(store: &'a mut StoreInner) -> Self {
428        let store_id = store.objects.id();
429        assert!(
430            !StoreContext::is_active(store_id) && !StoreContext::is_suspended(store_id),
431            "store is already on the current thread's context stack (active or suspended)"
432        );
433        StoreContext::push_cothread(store_id, NonNull::from(store));
434        Self {
435            store_id,
436            _store: std::marker::PhantomData,
437        }
438    }
439}
440
441#[cfg(feature = "unsafe-cothread")]
442impl Drop for CoroutineStoreGuard<'_> {
443    fn drop(&mut self) {
444        if std::thread::panicking() {
445            return;
446        }
447        StoreContext::uninstall_cothread(self.store_id);
448    }
449}

465impl Clone for StorePtrWrapper {
466    fn clone(&self) -> Self {
467        STORE_CONTEXT_STACK.with(|cell| {
468            let mut stack = cell.borrow_mut();
469            let top = stack
470                .last_mut()
471                .expect("No store context installed on this thread");
472            match unsafe { top.entry.get().as_ref().unwrap() } {
473                StoreContextEntry::Sync(ptr) if *ptr == self.store_ptr => (),
474                _ => panic!("Mismatched store context access"),
475            }
476            top.borrow_count += 1;
477            Self {
478                store_ptr: self.store_ptr,
479            }
480        })
481    }
482}

519impl Drop for StoreInstallGuard {
520    fn drop(&mut self) {
521        let Some(store_id) = self.store_id else {
522            return;
523        };
524        STORE_CONTEXT_STACK.with(|cell| {
525            let mut stack = cell.borrow_mut();
526            match (stack.pop(), std::thread::panicking()) {
527                (Some(top), false) => {
528                    assert_eq!(top.id, store_id, "Mismatched store context uninstall");
529                    assert_eq!(
530                        top.borrow_count, 0,
531                        "Cannot uninstall store context while it is still borrowed"
532                    );
533                }
534                (Some(top), true) => {
535                    // If we're panicking and there's a store ID mismatch, just
536                    // put the store back in the hope that its own install guard
537                    // take care of uninstalling it later.
538                    if top.id != store_id {
539                        stack.push(top);
540                    }
541                }
542                (None, false) => panic!("Store context stack underflow"),
543                (None, true) => {
544                    // Nothing to do if we're panicking; panics can put the context
545                    // in an invalid state, and we don't want to cause another panic here.
546                }
547            }
548        })
549    }
550}

605#[cfg(test)]
606mod borrow_provenance {
607    use super::*;
608    use crate::{AsStoreMut, Store};

610    /// Replays a recursive guest call at the level of the store context, which
611    /// is the part of it that has nothing to do with executing Wasm:
612    ///
613    /// ```text
614    /// Function::call(&mut store)          install(ptr from the caller's borrow)
615    ///   wasm -> import                    get_current            -> the shim's borrow
616    ///     shim: Function::call(&mut env)  install(ptr from *that* borrow)
617    ///       wasm -> import                get_current            -> the inner borrow
618    ///     shim carries on using its own borrow
619    /// ```
620    ///
621    /// Natively this always passes; it earns its keep under Miri, which is
622    /// where the last step is checked. If [`StoreContext::install`] ever goes
623    /// back to re-using an entry that is already active, the inner borrow
624    /// becomes a sibling of the shim's rather than a child of it, and this test
625    /// reports the shim's own store as invalidated:
626    ///
627    /// ```text
628    /// error: Undefined Behavior: trying to retag from <..> for Unique
629    ///        permission, but that tag does not exist in the borrow stack
630    ///   --> lib/api/src/entities/store/store_ref.rs   &mut self.inner.objects
631    /// ```
632    ///
633    /// Run it with:
634    ///
635    /// ```text
636    /// cargo +nightly miri test -p wasmer --features sys --lib borrow_provenance
637    /// ```
638    #[test]
639    fn nested_call_keeps_the_outer_borrow_usable() {
640        let mut store = Store::default();
641        let id = store.id();
642
643        // --- the embedder's Function::call(&mut store)
644        let mut caller = store.as_store_mut();
645        let install = unsafe { StoreContext::install(caller.as_store_mut().inner as *mut _) };
646
647        // --- the import trampoline, handing the shim its store
648        let mut wrapper = unsafe { StoreContext::get_current(id) };
649        let mut shim = wrapper.as_mut();
650        let _ = shim.objects_mut().id();
651
652        {
653            // --- the shim calls back into the guest
654            let inner_install =
655                unsafe { StoreContext::install(shim.as_store_mut().inner as *mut _) };
656            let pause = unsafe { StoreContext::pause(id) };
657
658            // --- the inner import trampoline
659            let mut inner_wrapper = unsafe { StoreContext::get_current(id) };
660            let mut inner_shim = inner_wrapper.as_mut();
661            let _ = inner_shim.objects_mut().id();
662            drop(inner_wrapper);
663
664            drop(pause);
665            drop(inner_install);
666        }
667
668        // --- and goes on using the borrow it held throughout
669        let _ = shim.objects_mut().id();
670
671        drop(wrapper);
672        drop(install);
673    }
674
675    /// The lending path, which is the nesting above with the inner entry
676    /// installed by [`StoreMut::parked`] instead of by a nested call: the
677    /// borrow handed out by [`crate::Store::with_current`] must not invalidate
678    /// the one the lender is holding. Miri checks the last line.
679    #[test]
680    fn a_lend_keeps_the_lending_borrow_usable() {
681        let mut store = Store::default();
682        let id = store.id();
683
684        let mut caller = store.as_store_mut();
685        let install = unsafe { StoreContext::install(caller.as_store_mut().inner as *mut _) };
686
687        let mut wrapper = unsafe { StoreContext::get_current(id) };
688        let mut shim = wrapper.as_mut();
689        let _ = shim.objects_mut().id();
690
691        shim.parked(|| {
692            crate::Store::with_current(|lent| {
693                let _ = lent.objects_mut().id();
694            })
695            .expect("a parked store is lendable");
696        });
697
698        // The lender goes back to the borrow it held throughout.
699        let _ = shim.objects_mut().id();
700
701        drop(wrapper);
702        drop(install);
703    }
